using System.Diagnostics;
using TSD;
internal static partial class Program
{
    // The 7 building (weather) arrays the TM59 ToSAM reads: annual calls vs one day-major pass of daily calls.
    internal static void WeatherRead(SimulationData sd, BuildingData bd)
    {
        tsdBuildingArray[] arrays = { tsdBuildingArray.cloudCover, tsdBuildingArray.diffuseRadiation, tsdBuildingArray.externalHumidity, tsdBuildingArray.externalTemperature, tsdBuildingArray.globalRadiation, tsdBuildingArray.windDirection, tsdBuildingArray.windSpeed };
        var s = Stopwatch.StartNew(); var ann = arrays.Select(a => F(bd.GetAnnualBuildingResult((int)a))).ToArray(); double tA = s.Elapsed.TotalSeconds;
        s.Restart(); var day = arrays.Select(_ => new float[8760]).ToArray();
        for (int d = sd.firstDay; d <= sd.lastDay; d++) for (int k = 0; k < arrays.Length; k++) Array.Copy(F(bd.GetDailyBuildingResult(d, (int)arrays[k])), 0, day[k], (d - 1) * 24, 24);
        double tD = s.Elapsed.TotalSeconds;
        int bad = Enumerable.Range(0, arrays.Length).Count(k => ann[k].Length != 8760 || !ann[k].Select(BitConverter.SingleToInt32Bits).SequenceEqual(day[k].Select(BitConverter.SingleToInt32Bits)));
        Log(string.Format("WEATHER 7 arrays: annual calls {0:0.0} s, day-major daily calls {1:0.0} s; arrays differing {2}; annual length {3}", tA, tD, bad, ann[0].Length));
    }
}
