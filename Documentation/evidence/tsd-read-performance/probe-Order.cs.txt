using TSD;
internal static partial class Program
{
    // Does TAS sum a group in the order given, or in its own zone order? Reverse and shuffle the members.
    internal static void OrderCheck(SimulationData sd, BuildingData bd, List<ZoneData> zones, float[][] cl, List<(string, int[])> groups)
    {
        var rnd = new Random(1); int n = 0, given = 0, building = 0, invariant = 0;
        foreach (var (label, m0) in groups)
        {
            if (m0.Length < 3 || m0.Any(i => i < 0)) continue;
            foreach (int[] m in new[] { m0.Reverse().ToArray(), m0.OrderBy(_ => rnd.Next()).ToArray() })
            {
                object[,] t = (object[,])bd.GetPeakZoneGroupGains(new[] { "Name:" + string.Join(":", m.Select(i => zones[i].zoneGUID)) }, new[] { (int)tsdZoneArray.coolingLoad });
                double tv = System.Convert.ToDouble(t[1, 0]); int ti = System.Convert.ToInt32(t[2, 0]);
                (float, int) P(int[] order) { float b = 0; int bi = 0; for (int h = 0; h < cl[0].Length; h++) { float s = 0; foreach (int i in order) s += cl[i][h]; if (s > b) { b = s; bi = h + 1; } } return (b, bi); }
                var pg = P(m); var pb = P(m.OrderBy(i => i).ToArray());
                bool g = pg.Item1 == tv && pg.Item2 == ti, b = pb.Item1 == tv && pb.Item2 == ti; n++;
                if (g) given++; if (b) building++; if (pg == pb) invariant++;
            }
        }
        Log(string.Format("ORDER: {0} permuted multi-zone queries: given-order sum exact {1}, building-order sum exact {2}; order-invariant sums {3}", n, given, building, invariant));
    }
}
