// 2026-09-29 exchanger-representation investigation. NOT production code. Operates on COPIES only.
//   exchprobe <tpd> <outdir> [variants=V1,V2,DT]
// BASE = the untouched copy. V1 = [2D ODB x EDB2 recovery indicator (Equal), 1D EFlow EF..CEF (Multiply)] on the
// breakpoint-pruned axes. V2 = same two tables in the opposite order (1D Equal first, 2D Multiply second).
// DT = a 1D table over tpdProfileDataVariableDeltaT mapping [-50, 50] -> [0, 1] (diagnostic: what IS DeltaT?).
// Every copy is plant-room simulated with duct data (production GuidanceCoolingResults flags) and every duct's
// flow (1) and temperature (3) series is compared bit-for-bit with BASE.
using System;
using System.Collections;
using System.Collections.Generic;
using System.Diagnostics;
using System.Globalization;
using System.IO;
using System.Linq;
using System.Text;
using SAM.Core.Tas;

namespace P0
{
    internal static class Exch
    {
        private static readonly CultureInfo ci = CultureInfo.InvariantCulture;
        private static StringBuilder sb;
        private static void W(string format, params object[] a) { string l = a.Length == 0 ? format : string.Format(ci, format, a); l = string.Format(ci, "[{0:HH:mm:ss}] {1}", DateTime.Now, l); Console.WriteLine(l); sb.AppendLine(l); }

        //The real project's recipe (MG-Review: bypass intake >= 12, extract > intake and >= 19; 0.8 / 0.8576; act 22).
        private const double Imin = 12.0, Emin = 19.0, Act = 22.0, EF = 0.8, CEF = 0.8576;

        private static double[] Axis(double[] low, double from, double to, double[] high, double[] extra)
        {
            SortedSet<double> values = new SortedSet<double>();
            foreach (double v in low) if (v < from - 1e-9) values.Add(v);
            int steps = (int)Math.Round((to - from) / 0.1);
            for (int i = 0; i <= steps; i++) values.Add(Math.Round(from + (0.1 * i), 4));
            foreach (double v in high) values.Add(v);
            foreach (double v in extra) values.Add(Math.Round(v, 4));
            return values.ToArray();
        }
        private static bool Bypass(double i, double e) => i >= Imin && e > i && e >= Emin;

        internal static double[] Intakes() => Axis(new double[] { -20, -5, 5 }, Math.Min(Imin, 12.0), 45.0, new double[0], new double[] { Imin - 0.01 });
        internal static double[] Extracts() => Axis(new double[] { 5, 12 }, Math.Min(Emin, 18.0) - 0.1, 45.0, new double[] { 45.1, 50, 60, 80, 100 }, new double[] { Emin - 0.01, Act + 0.01 });

        /// <summary>Keeps a breakpoint unless its whole line equals BOTH neighbour lines (then it is collinear: removable exactly).</summary>
        internal static int[] Keep(int n, Func<int, int, bool> same)
        {
            List<int> k = new List<int> { 0 };
            for (int x = 1; x < n - 1; x++) if (!(same(x, x - 1) && same(x, x + 1))) k.Add(x);
            k.Add(n - 1);
            return k.ToArray();
        }

        /// <summary>tpdcmp &lt;before.tpd&gt; &lt;after.tpd&gt; &lt;outdir&gt;: COPIES of both simulated with duct data; every duct series compared.</summary>
        public static int Compare(string[] args)
        {
            string dir = args[3];
            Directory.CreateDirectory(dir);
            sb = new StringBuilder();
            Dictionary<string, float[]>[] results = new Dictionary<string, float[]>[2];
            for (int n = 0; n < 2; n++)
            {
                string copy = Path.Combine(dir, "tpdcmp-" + (n == 0 ? "before" : "after") + ".tpd");
                File.Copy(args[1 + n], copy, true);
                using (SAMTPDDocument doc = new SAMTPDDocument(copy))
                {
                    object ec = doc.TPDDocument.EnergyCentre;
                    Dictionary<string, float[]> series = new Dictionary<string, float[]>(StringComparer.Ordinal);
                    foreach (object system in Graph.Systems(ec))
                    {
                        foreach (object component in Graph.TopLevel(system))
                        {
                            if (!(component is global::TPD.Exchanger exchanger)) continue;
                            dynamic se = exchanger.SensibleEfficiency;
                            List<string> m = new List<string>();
                            for (int k = 1; k <= (int)se.GetModifierCount(); k++)
                            {
                                dynamic t = se.GetModifier(k);
                                m.Add(string.Format(ci, "[{0}: {1}x{2}x{3} mult={4} vars={5}/{6}/{7}]", k, (int)t.GetAxisSize(1), (int)t.GetAxisSize(2), (int)t.GetAxisSize(3), t.Multiplier, t.GetVariable(1), (int)t.GetAxisSize(2) > 0 ? t.GetVariable(2) : "-", (int)t.GetAxisSize(3) > 0 ? t.GetVariable(3) : "-"));
                            }
                            W("{0} {1}: efficiency {2}, {3} modifier(s) {4}", n == 0 ? "BEFORE" : "AFTER ", Dump.Get(system, "Name"), (double)se.Value, (int)se.GetModifierCount(), string.Join(" ", m));
                        }
                    }

                    double pollutant = (double)((dynamic)ec).ExternalPollutant.Value;
                    for (int p = 1; p <= Graph.Int(Dump.Call(ec, "GetPlantRoomCount")); p++)
                    {
                        dynamic plantRoom = Dump.Call(ec, "GetPlantRoom", p);
                        W("   plant room {0}: \"{1}\"", p, ((string)plantRoom.SimulateEx(1, 8760, 0, pollutant, 10.0, 47, 1, 0) ?? "").Trim());
                    }

                    foreach (object system in Graph.Systems(ec))
                    {
                        string sn = Dump.Get(system, "Name");
                        List<object> ducts = Graph.Ducts(system);
                        for (int d = 0; d < ducts.Count; d++)
                        {
                            foreach (int v in new[] { 1, 3 })
                            {
                                IEnumerable e = ((dynamic)ducts[d]).GetResultsData(global::TPD.tpdResultsPeriod.tpdResultsPeriodHourly, global::TPD.tpdCombinerType.tpdCombinerTypeMax, v, 1, 8760) as IEnumerable;
                                if (e == null) continue;
                                series[System.Text.RegularExpressions.Regex.Replace(string.Format(ci, "{0}|duct{1:000} {2}|v{3}", sn, d, Graph.Edge(ducts[d]), v), @"\{[0-9A-Fa-f-]{36}\}", "{guid}")] = e.Cast<object>().Select(o => System.Convert.ToSingle(o)).ToArray();
                            }
                        }
                    }
                    results[n] = series;
                }
            }

            int identical = 0; double maxd = 0;
            foreach (string key in results[0].Keys)
            {
                if (!results[1].TryGetValue(key, out float[] y)) { W("missing AFTER {0}", key); continue; }
                double d = 0; for (int i = 0; i < y.Length; i++) d = Math.Max(d, Math.Abs(y[i] - results[0][key][i]));
                if (d == 0 && y.Length == results[0][key].Length) identical++;
                maxd = Math.Max(maxd, d);
            }
            W("TPDCMP: {0} of {1} duct series bit-identical (AFTER has {2}); max |diff| {3:G6}", identical, results[0].Count, results[1].Count, maxd);
            File.WriteAllText(Path.Combine(dir, "tpdcmp.txt"), sb.ToString());
            return 0;
        }

        public static int Run(string[] args)
        {
            string source = args[1], dir = args[2];
            string[] variants = (args.Skip(3).FirstOrDefault(x => x.StartsWith("variants=")) ?? "variants=V1,V2,DT").Substring(9).Split(',');
            Directory.CreateDirectory(dir);
            sb = new StringBuilder();

            double[] I = Intakes(), E = Extracts();
            int[] ki = Keep(I.Length, (a, b) => E.All(e => Bypass(I[a], e) == Bypass(I[b], e)));
            int[] kj = Keep(E.Length, (a, b) => I.All(i => Bypass(i, E[a]) == Bypass(i, E[b])));
            double[] PI = ki.Select(x => I[x]).ToArray(), PE = kj.Select(x => E[x]).ToArray();
            W("full axes {0} x {1}; pruned {2} x {3} = {4} cells", I.Length, E.Length, PI.Length, PE.Length, PI.Length * PE.Length);

            Dictionary<string, Dictionary<string, float[]>> results = new Dictionary<string, Dictionary<string, float[]>>();
            foreach (string variant in new[] { "BASE" }.Concat(variants))
            {
                string copy = Path.Combine(dir, "exchprobe-" + variant + ".tpd");
                File.Copy(source, copy, true);
                using (SAMTPDDocument doc = new SAMTPDDocument(copy))
                {
                    object ec = doc.TPDDocument.EnergyCentre;
                    foreach (object system in Graph.Systems(ec))
                    {
                        foreach (object component in Graph.TopLevel(system))
                        {
                            if (!(component is global::TPD.Exchanger exchanger)) continue;
                            dynamic se = exchanger.SensibleEfficiency;
                            dynamic old = se.GetModifier(1);
                            double design = (double)old.GetAxisValue(3, 1), elevated = (double)old.GetAxisValue(3, 2);
                            W("{0} {1}: value={2} modifiers={3} old table {4}x{5}x{6} flow {7}/{8}", variant, Dump.Get(system, "Name"), (double)se.Value, (int)se.GetModifierCount(), (int)old.GetAxisSize(1), (int)old.GetAxisSize(2), (int)old.GetAxisSize(3), design, elevated);
                            if (variant == "BASE") continue;

                            Stopwatch sw = Stopwatch.StartNew();
                            long writes = 0, reads = 0;
                            se.ClearModifiers();
                            if (variant == "V1" || variant == "V2")
                            {
                                dynamic first = se.AddModifierTable();
                                dynamic second = se.AddModifierTable();
                                dynamic grid = variant == "V1" ? first : second;
                                dynamic flow = variant == "V1" ? second : first;

                                grid.SetVariable(1, global::TPD.tpdProfileDataVariableType.tpdProfileDataVariableODB);
                                grid.SetVariable(2, global::TPD.tpdProfileDataVariableType.tpdProfileDataVariableEDB2);
                                grid.SetSize(PI.Length, PE.Length, 0);
                                grid.Extrapolate = false;
                                grid.Multiplier = variant == "V1" ? global::TPD.tpdProfileDataModifierMultiplier.tpdProfileDataModifierEqual : global::TPD.tpdProfileDataModifierMultiplier.tpdProfileDataModifierMultiply;
                                W("   grid after SetSize({0},{1},0): axis sizes {2} x {3} x {4}", PI.Length, PE.Length, (int)grid.GetAxisSize(1), (int)grid.GetAxisSize(2), (int)grid.GetAxisSize(3));
                                for (int i = 0; i < PI.Length; i++) { grid.SetAxisValue(1, i + 1, PI[i]); writes++; }
                                for (int j = 0; j < PE.Length; j++) { grid.SetAxisValue(2, j + 1, PE[j]); writes++; }
                                for (int i = 0; i < PI.Length; i++) for (int j = 0; j < PE.Length; j++) { grid.SetDataValue(i + 1, j + 1, 1, Bypass(PI[i], PE[j]) ? 0.0 : 1.0); writes++; }

                                flow.SetVariable(1, global::TPD.tpdProfileDataVariableType.tpdProfileDataVariableEFlow);
                                flow.SetSize(2, 0, 0);
                                flow.Extrapolate = false;
                                flow.Multiplier = variant == "V1" ? global::TPD.tpdProfileDataModifierMultiplier.tpdProfileDataModifierMultiply : global::TPD.tpdProfileDataModifierMultiplier.tpdProfileDataModifierEqual;
                                flow.SetAxisValue(1, 1, design); flow.SetAxisValue(1, 2, elevated);
                                flow.SetDataValue(1, 1, 1, EF); flow.SetDataValue(2, 1, 1, CEF);
                                writes += 4;
                                W("   wrote {0} calls in {1:0.0} s", writes, sw.Elapsed.TotalSeconds);

                                sw.Restart();
                                int bad = 0;
                                for (int i = 0; i < PI.Length; i++) { if ((double)grid.GetAxisValue(1, i + 1) != PI[i]) bad++; reads++; }
                                for (int j = 0; j < PE.Length; j++) { if ((double)grid.GetAxisValue(2, j + 1) != PE[j]) bad++; reads++; }
                                for (int i = 0; i < PI.Length; i++) for (int j = 0; j < PE.Length; j++) { if ((double)grid.GetDataValue(i + 1, j + 1, 1) != (Bypass(PI[i], PE[j]) ? 0.0 : 1.0)) bad++; reads++; }
                                W("   read back {0} calls in {1:0.0} s, {2} disagreements; modifier count {3}, types {4}/{5}, flow cells {6}/{7}", reads, sw.Elapsed.TotalSeconds, bad, (int)se.GetModifierCount(), se.GetModifierType(1), se.GetModifierType(2), (double)flow.GetDataValue(1, 1, 1), (double)flow.GetDataValue(2, 1, 1));
                            }
                            else if (variant == "DT")
                            {
                                dynamic t = se.AddModifierTable();
                                t.SetVariable(1, global::TPD.tpdProfileDataVariableType.tpdProfileDataVariableDeltaT);
                                t.SetSize(2, 0, 0);
                                t.Extrapolate = false;
                                t.Multiplier = global::TPD.tpdProfileDataModifierMultiplier.tpdProfileDataModifierEqual;
                                t.SetAxisValue(1, 1, -50.0); t.SetAxisValue(1, 2, 50.0);
                                t.SetDataValue(1, 1, 1, 0.0); t.SetDataValue(2, 1, 1, 1.0);
                            }
                        }
                    }

                    W("{0}: simulating", variant);
                    Stopwatch sim = Stopwatch.StartNew();
                    double pollutant = (double)((dynamic)ec).ExternalPollutant.Value;
                    for (int p = 1; p <= Graph.Int(Dump.Call(ec, "GetPlantRoomCount")); p++)
                    {
                        dynamic plantRoom = Dump.Call(ec, "GetPlantRoom", p);
                        string diag = (string)plantRoom.SimulateEx(1, 8760, 0, pollutant, 10.0, 47, 1, 0);
                        W("   plant room {0}: \"{1}\" ({2:0.0} s)", p, (diag ?? "").Trim(), sim.Elapsed.TotalSeconds);
                    }

                    Dictionary<string, float[]> series = new Dictionary<string, float[]>(StringComparer.Ordinal);
                    foreach (object system in Graph.Systems(ec))
                    {
                        string sn = Dump.Get(system, "Name");
                        List<object> ducts = Graph.Ducts(system);
                        for (int d = 0; d < ducts.Count; d++)
                        {
                            foreach (int v in new[] { 1, 3 })
                            {
                                IEnumerable e = ((dynamic)ducts[d]).GetResultsData(global::TPD.tpdResultsPeriod.tpdResultsPeriodHourly, global::TPD.tpdCombinerType.tpdCombinerTypeMax, v, 1, 8760) as IEnumerable;
                                if (e == null) continue;
                                series[string.Format(ci, "{0}|duct{1:000} {2}|v{3}", sn, d, Graph.Edge(ducts[d]), v)] = e.Cast<object>().Select(o => System.Convert.ToSingle(o)).ToArray();
                            }
                        }
                    }
                    results[variant] = series;
                    W("{0}: {1} series read", variant, series.Count);
                    //never saved - the copy is discarded
                }
            }

            Dictionary<string, float[]> b = results["BASE"];
            foreach (string variant in variants.Where(v => v != "DT"))
            {
                Dictionary<string, float[]> x = results[variant];
                int identical = 0; double maxd = 0; long hours = 0; string worst = null;
                foreach (string key in b.Keys)
                {
                    if (!x.TryGetValue(key, out float[] y)) { W("{0}: missing {1}", variant, key); continue; }
                    double d = 0; int h = 0;
                    for (int i = 0; i < Math.Min(y.Length, b[key].Length); i++) { double dd = Math.Abs(y[i] - b[key][i]); if (dd > 0) h++; d = Math.Max(d, dd); }
                    if (d == 0 && y.Length == b[key].Length) identical++; else { hours += h; if (d > maxd) { maxd = d; worst = key; } }
                }
                W("COMPARE {0} vs BASE: {1} of {2} duct series bit-identical; max |diff| {3:G6}; differing series-hours {4}; worst {5}", variant, identical, b.Count, maxd, hours, worst);
            }

            if (results.ContainsKey("DT"))
            {
                //For each exchanger, infer the efficiency TAS applied from its ducts, then DeltaT = 100 * eff - 50.
                Dictionary<string, float[]> x = results["DT"];
                foreach (string key in x.Keys.Where(k => k.Contains("[Exchanger") && k.EndsWith("|v3")).OrderBy(k => k).Take(12)) W("  DT duct {0}", key);
                File.WriteAllText(Path.Combine(dir, "exchprobe-DT-keys.txt"), string.Join(Environment.NewLine, x.Keys));
                using (StreamWriter w = new StreamWriter(Path.Combine(dir, "exchprobe-DT-series.csv")))
                {
                    List<string> keys = x.Keys.Where(k => k.Contains("Exchanger") && k.EndsWith("|v3")).OrderBy(k => k).ToList();
                    w.WriteLine("hour," + string.Join(",", keys.Select(k => "\"" + k + "\"")));
                    for (int h = 0; h < 8760; h++) w.WriteLine(h + "," + string.Join(",", keys.Select(k => x[k][h].ToString("R", ci))));
                }
            }

            File.WriteAllText(Path.Combine(dir, "exchprobe.txt"), sb.ToString());
            return 0;
        }
    }
